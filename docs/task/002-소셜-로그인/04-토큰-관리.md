# 04. 토큰 관리 (SecureStore)

## 목표

`expo-secure-store`를 사용하여 JWT 토큰을 안전하게 저장/조회/삭제합니다.

## 상태

⬜ 대기

## 왜 SecureStore인가?

| 저장소 | 보안 수준 | 사용 |
|--------|----------|------|
| AsyncStorage | 낮음 (평문) | 비민감 데이터 |
| **SecureStore** | 높음 (암호화) | **토큰, 민감 정보** |
| MMKV | 중간 | 대용량 데이터 |

SecureStore는 내부적으로:
- **iOS**: Keychain 사용 (Secure Enclave 보호)
- **Android**: Keystore 사용 (TEE 보호)

### 2. 토큰 저장소 구현

#### src/lib/tokenStorage.ts

```typescript
import * as SecureStore from 'expo-secure-store';

// 저장소 키 상수
const STORAGE_KEYS = {
  ACCESS_TOKEN: 'accessToken',
  REFRESH_TOKEN: 'refreshToken',
  USER_INFO: 'userInfo',
} as const;

// 토큰 타입
export interface Tokens {
  accessToken: string;
  refreshToken: string;
}

// 사용자 정보 타입
export interface UserInfo {
  id: string;
  email: string;
  name: string;
  profileImage?: string;
}

// ============ 토큰 저장 ============

export const saveTokens = async (tokens: Tokens): Promise<void> => {
  await Promise.all([
    SecureStore.setItemAsync(STORAGE_KEYS.ACCESS_TOKEN, tokens.accessToken),
    SecureStore.setItemAsync(STORAGE_KEYS.REFRESH_TOKEN, tokens.refreshToken),
  ]);
};

// ============ 토큰 조회 ============

export const getAccessToken = async (): Promise<string | null> => {
  return SecureStore.getItemAsync(STORAGE_KEYS.ACCESS_TOKEN);
};

export const getRefreshToken = async (): Promise<string | null> => {
  return SecureStore.getItemAsync(STORAGE_KEYS.REFRESH_TOKEN);
};

export const getTokens = async (): Promise<Tokens | null> => {
  const [accessToken, refreshToken] = await Promise.all([
    getAccessToken(),
    getRefreshToken(),
  ]);

  if (!accessToken || !refreshToken) {
    return null;
  }

  return { accessToken, refreshToken };
};

// ============ 토큰 삭제 ============

export const clearTokens = async (): Promise<void> => {
  await Promise.all([
    SecureStore.deleteItemAsync(STORAGE_KEYS.ACCESS_TOKEN),
    SecureStore.deleteItemAsync(STORAGE_KEYS.REFRESH_TOKEN),
  ]);
};

// ============ 사용자 정보 ============

export const saveUserInfo = async (user: UserInfo): Promise<void> => {
  await SecureStore.setItemAsync(
    STORAGE_KEYS.USER_INFO,
    JSON.stringify(user)
  );
};

export const getUserInfo = async (): Promise<UserInfo | null> => {
  const data = await SecureStore.getItemAsync(STORAGE_KEYS.USER_INFO);
  return data ? JSON.parse(data) : null;
};

export const clearUserInfo = async (): Promise<void> => {
  await SecureStore.deleteItemAsync(STORAGE_KEYS.USER_INFO);
};

// ============ 전체 삭제 (로그아웃) ============

export const clearAll = async (): Promise<void> => {
  await Promise.all([clearTokens(), clearUserInfo()]);
};

// ============ 인증 상태 확인 ============

export const isAuthenticated = async (): Promise<boolean> => {
  const token = await getAccessToken();
  return token !== null;
};
```

### 3. 토큰 유효성 검사 유틸리티

#### src/lib/tokenUtils.ts

```typescript
// JWT 디코딩 (검증 없이 페이로드만 추출)
export const decodeJWT = (token: string): Record<string, unknown> | null => {
  try {
    const base64Url = token.split('.')[1];
    const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    const jsonPayload = decodeURIComponent(
      atob(base64)
        .split('')
        .map((c) => '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2))
        .join('')
    );
    return JSON.parse(jsonPayload);
  } catch {
    return null;
  }
};

// 토큰 만료 확인
export const isTokenExpired = (token: string): boolean => {
  const payload = decodeJWT(token);
  if (!payload || typeof payload.exp !== 'number') {
    return true;
  }

  // 만료 10초 전부터 만료로 판단 (버퍼)
  const now = Math.floor(Date.now() / 1000);
  return payload.exp < now + 10;
};

// 토큰 만료까지 남은 시간 (초)
export const getTokenRemainingTime = (token: string): number => {
  const payload = decodeJWT(token);
  if (!payload || typeof payload.exp !== 'number') {
    return 0;
  }

  const now = Math.floor(Date.now() / 1000);
  return Math.max(0, payload.exp - now);
};
```

### 4. 인증 상태 관리 Hook

#### src/lib/useAuth.ts

```typescript
import { useState, useEffect, useCallback } from 'react';
import {
  getTokens,
  saveTokens,
  clearAll,
  getUserInfo,
  saveUserInfo,
  Tokens,
  UserInfo,
} from './tokenStorage';
import { isTokenExpired } from './tokenUtils';

interface AuthState {
  isLoading: boolean;
  isAuthenticated: boolean;
  user: UserInfo | null;
  tokens: Tokens | null;
}

interface AuthActions {
  login: (tokens: Tokens, user: UserInfo) => Promise<void>;
  logout: () => Promise<void>;
  refreshAuth: () => Promise<void>;
}

export const useAuth = (): AuthState & AuthActions => {
  const [state, setState] = useState<AuthState>({
    isLoading: true,
    isAuthenticated: false,
    user: null,
    tokens: null,
  });

  // 인증 상태 초기화
  useEffect(() => {
    const initAuth = async () => {
      try {
        const [tokens, user] = await Promise.all([
          getTokens(),
          getUserInfo(),
        ]);

        if (tokens && !isTokenExpired(tokens.accessToken)) {
          setState({
            isLoading: false,
            isAuthenticated: true,
            user,
            tokens,
          });
        } else {
          // 토큰이 없거나 만료됨
          await clearAll();
          setState({
            isLoading: false,
            isAuthenticated: false,
            user: null,
            tokens: null,
          });
        }
      } catch {
        setState({
          isLoading: false,
          isAuthenticated: false,
          user: null,
          tokens: null,
        });
      }
    };

    initAuth();
  }, []);

  // 로그인
  const login = useCallback(async (tokens: Tokens, user: UserInfo) => {
    await Promise.all([saveTokens(tokens), saveUserInfo(user)]);
    setState({
      isLoading: false,
      isAuthenticated: true,
      user,
      tokens,
    });
  }, []);

  // 로그아웃
  const logout = useCallback(async () => {
    await clearAll();
    setState({
      isLoading: false,
      isAuthenticated: false,
      user: null,
      tokens: null,
    });
  }, []);

  // 인증 상태 새로고침
  const refreshAuth = useCallback(async () => {
    const [tokens, user] = await Promise.all([getTokens(), getUserInfo()]);
    setState((prev) => ({
      ...prev,
      tokens,
      user,
      isAuthenticated: tokens !== null,
    }));
  }, []);

  return {
    ...state,
    login,
    logout,
    refreshAuth,
  };
};
```

## 사용 예시

### 로그인 후 토큰 저장

```typescript
import { saveTokens, saveUserInfo } from '@lib/tokenStorage';

const handleLoginSuccess = async (response: AuthResponse) => {
  await saveTokens({
    accessToken: response.accessToken,
    refreshToken: response.refreshToken,
  });

  await saveUserInfo(response.user);
};
```

### API 요청 시 토큰 사용

```typescript
import { getAccessToken } from '@lib/tokenStorage';

const fetchUserData = async () => {
  const token = await getAccessToken();

  const response = await fetch('/api/user', {
    headers: {
      Authorization: `Bearer ${token}`,
    },
  });

  return response.json();
};
```

### 로그아웃

```typescript
import { clearAll } from '@lib/tokenStorage';

const handleLogout = async () => {
  await clearAll();
  router.replace('/(auth)/login');
};
```
